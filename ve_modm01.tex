\begin{multicols}{2}
\includegraphics[width=8.25cm]{images/viscoelasticity/modm01b} \\
\[
\dot{\bm \varepsilon} = \frac12 (\vec\nabla \vec\upnu  + \vec\nabla \vec\upnu^T  ) 
\]

\[
\dot{\bm \varepsilon}_{\rm e}^d 
=
\dot{\bm \varepsilon}_{\rm e} - 
 \frac13 {\rm tr}(\dot{\bm \varepsilon}_{\rm e}) {\bm 1} 
\]
\[
\dot{\bm \varepsilon}_{\rm v}^d 
=
\dot{\bm \varepsilon}_{\rm v} 
- \frac13 {\rm tr}(\dot{\bm \varepsilon}_{\rm v}) {\bm 1} 
\]
with
\[
\dot{\bm \varepsilon} = \dot{\bm \varepsilon}_{\rm e} + \dot{\bm \varepsilon}_{\rm v} 
\]
which leads to 
\[
\dot{\bm \varepsilon}^d = \dot{\bm \varepsilon}_{\rm e}^d + \dot{\bm \varepsilon}_{\rm v}^d 
\]
and
\[
{\rm tr}(\dot{\bm \varepsilon}) = 
{\rm tr}(\dot{\bm \varepsilon}_{\rm e}) + 
{\rm tr}(\dot{\bm \varepsilon}_{\rm v}) 
\]


\end{multicols}

\begin{multicols}{2}
\includegraphics[width=8.25cm]{images/viscoelasticity/modm01c} \\
\[
\frac{ \check{\bm \tau}}{2\mu} + \frac{{\bm \tau}}{2\eta}  
= \dot{\bm \varepsilon}_{\rm e}^d + \dot{\bm \varepsilon}_{\rm v}^d = \dot{\bm \varepsilon}^d 
\]
$\check{\bm \tau}$ is the Jaumann corotational stress rate
\end{multicols}

\[
\frac{\dot{p}}{K_{\rm e}}+\frac{p}{\xi} = - {\rm tr} (\dot{\bm \varepsilon})
\]
Note that $\check{p}=\dot{p}$ as it is a scalar.

\[
\check{\bm \tau} = \dot{\bm \tau} + {\bm \tau} \cdot \dot{\bm \omega} - \dot{\bm \omega} \cdot {\bm \tau}  
\]
\[
\dot{\bm \omega} = \frac12 (\vec\nabla \vec\upnu  - \vec\nabla \vec\upnu^T  ) 
\]
where $\dot{\bm\omega}$ is the material spin tensor.
\end{multicols}
