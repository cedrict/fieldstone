\[
\tau_{ij}^{k+1} = 2 \eta_{ve} \dot{\varepsilon}_{ij} + \left(1+\frac{G \delta t}{\eta}\right)^{-1} 
( \tau_{ij}^k - \tau^k_{ik} \omega_{kj} - \omega_{ik} \tau{kj}^k)
\]

\[
\eta_{ve}= \left(  \frac{1}{\eta} + \frac{1}{G \delta t} \right)^{-1}
\]
